import { test, type Page, type CDPSession } from "@playwright/test";
import { writeFileSync } from "node:fs";
import type { SetlistItem } from "../../src/types/setList";
import { signInAs, musicien, type FakeProfile } from "../helpers/fakeSession";
import { attendreCompteur, ouvrirMenu } from "../helpers/louange";
import { prepare, mesure, LARGEUR } from "./outil";

test.use({ viewport: { width: LARGEUR, height: 900 } });
const DOSSIER = "/tmp/claude-0/-home-user/d2681d54-ea1e-5437-a805-4ba46328d8b8/scratchpad/profils";
const ID = "setlist-mesure";
const PROFIL: FakeProfile = { uid: "uid-sacha", email: "sacha@example.com", firstName: "Sacha", lastName: "Quérel", serviceRoles: { "Culte Francophone": ["musicien"] }, ...musicien("piano") };
const item = (songSlug: string, position: number): SetlistItem => ({ songSlug, position, keyOverride: null, showChords: true, showPinyin: true, useJianpu: false, structureOverride: null, sectionNotes: {}, notes: "" } as SetlistItem);
const SETLIST = { title: "Culte de mesure", leader: "Présidence", category: "Culte Francophone", date: "2026-11-29", language: "mixed", notes: "", ownerId: "uid-sacha", isPrivate: false,
  items: [item("abba-pere", 1), item("beni-soit-ton-nom", 2), item("一生爱你", 3), item("为我而来", 4)] };

async function profile(page: Page, cdp: CDPSession, nom: string, geste: () => Promise<unknown>, pret?: () => Promise<unknown>) {
  await cdp.send("Profiler.enable");
  await cdp.send("Profiler.setSamplingInterval", { interval: 250 });
  const c0 = (await cdp.send("Performance.getMetrics")).metrics.find((m) => m.name === "LayoutCount")!.value;
  await cdp.send("Profiler.start");
  await mesure(page, cdp, `Profil ${LARGEUR}`, nom, geste, pret);
  const { profile: p } = await cdp.send("Profiler.stop");
  const c1 = (await cdp.send("Performance.getMetrics")).metrics.find((m) => m.name === "LayoutCount")!.value;
  writeFileSync(`${DOSSIER}/${LARGEUR}-${nom}.cpuprofile`, JSON.stringify(p));
  console.log(`LayoutCount ${nom}: ${c1 - c0}`);
}

test("profils", async ({ page }) => {
  const cdp = await prepare(page);
  await signInAs(page, PROFIL, { [`setlists/${ID}`]: SETLIST }, `/setlists/${ID}`);
  const louange = page.getByRole("button", { name: /Mode Louange/ }).filter({ visible: true }).first();
  await louange.waitFor({ timeout: 120_000 });
  await page.waitForTimeout(3000);
  const crayon = page.getByRole("button", { name: "Crayon", exact: true }).filter({ visible: true }).first();
  await profile(page, cdp, "crayon-on", () => crayon.click(), () => page.locator("[data-barre-crayon]").waitFor());
  await profile(page, cdp, "crayon-off", () => page.locator("[data-barre-crayon]").getByRole("button", { name: "Terminé" }).click(), () => page.locator("[data-barre-crayon]").waitFor({ state: "detached" }));
  await profile(page, cdp, "louange", () => louange.click(), () => attendreCompteur(page, /^\d+ \/ \d+$/, 120_000));
  const marge = page.locator('[data-testid="marge-affichage"][data-mode="louange"]');
  await ouvrirMenu(page, "affichage");
  await marge.getByRole("radio", { name: "Plan" }).click();
  await page.waitForTimeout(2000);
  await profile(page, cdp, "plan-vers-tout", () => marge.getByRole("radio", { name: "Tout le chant" }).click());
});
