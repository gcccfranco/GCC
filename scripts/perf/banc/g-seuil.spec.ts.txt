test("seuil de largeur", async ({ page }) => {
  const cdp = await prepare(page);
  await signInAs(page, PROFIL, { [`setlists/${ID}`]: SETLIST }, `/setlists/${ID}`);
  const louange = page.getByRole("button", { name: /Mode Louange/ }).filter({ visible: true }).first();
  await louange.waitFor({ timeout: 60_000 });
  await page.waitForTimeout(2500);
  const etat = await page.evaluate(() => ({
    lignes: document.querySelectorAll("[data-rangee]").length,
    noeuds: document.getElementsByTagName("*").length,
    barre: Math.round(document.querySelector("aside, nav[aria-label]")?.getBoundingClientRect().width ?? 0),
    vue: Math.round((document.querySelector("[data-vue]") as HTMLElement | null)?.getBoundingClientRect().width ?? 0),
    item: Math.round(document.querySelector("[data-outline-item]")?.getBoundingClientRect().width ?? 0),
    colonnes: [...document.querySelectorAll("[data-outline-item] *")].filter((e) => { const c = getComputedStyle(e).columnCount; return c !== "auto" && c !== "1"; }).length,
  }));

  const lc = async () => (await cdp.send("Performance.getMetrics")).metrics.find((m) => m.name === "LayoutCount")!.value;

  let a = await lc();
  await mesure(page, cdp, "Seuil", "activer le Crayon", () => crayon.click(), () => page.locator("[data-barre-crayon]").waitFor());
  let b = await lc(); console.log("LAYOUTS crayon-on " + (b - a)); a = b;
  await mesure(page, cdp, "Seuil", "quitter le Crayon", () => page.locator("[data-barre-crayon]").getByRole("button", { name: "Terminé" }).click(), () => page.locator("[data-barre-crayon]").waitFor({ state: "detached" }));
  b = await lc(); console.log("LAYOUTS crayon-off " + (b - a)); a = b;
  await mesure(page, cdp, "Seuil", "lancer le mode louange", () => louange.click(), () => pret(page));
  b = await lc(); console.log("LAYOUTS louange " + (b - a));
});
