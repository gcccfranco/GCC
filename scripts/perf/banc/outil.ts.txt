import { type Page, type CDPSession, type Locator } from "@playwright/test";
import { appendFileSync } from "node:fs";

export const LARGEUR = Number(process.env.MESURE_LARGEUR ?? 1280);
export const RALENTI = Number(process.env.MESURE_CPU ?? (LARGEUR <= 400 ? 4 : 1));
export const BASE = process.env.MESURE_BASE ?? "http://localhost:3300";
export const MODE = process.env.MESURE_MODE ?? "construit";
const SORTIE = process.env.MESURE_SORTIE ?? "/tmp/claude-0/-home-user/d2681d54-ea1e-5437-a805-4ba46328d8b8/scratchpad/mesures.jsonl";

/** Script posé avant toute page : chrono (premier geste), tâches longues, dernières mutations du DOM. */
export const INIT = () => {
  const w = window as unknown as Record<string, any>;
  w.__m = { t0: 0, long: 0, nLong: 0, lastLong: 0, lastMut: 0 };
  const go = (e: Event) => { if (w.__m.arme && !w.__m.t0) w.__m.t0 = e.timeStamp; };
  for (const t of ["pointerdown", "keydown", "mousedown"]) window.addEventListener(t, go, { capture: true });
  try {
    new PerformanceObserver((l) => { for (const e of l.getEntries()) { w.__m.long += e.duration; w.__m.nLong++; w.__m.lastLong = Math.max(w.__m.lastLong, e.startTime + e.duration); } }).observe({ type: "longtask", buffered: true });
  } catch {}
  const mo = () => new MutationObserver(() => { w.__m.lastMut = performance.now(); }).observe(document, { subtree: true, childList: true, attributes: true, characterData: true });
  if (document.documentElement) mo(); else document.addEventListener("readystatechange", mo, { once: true });
};

export async function prepare(page: Page): Promise<CDPSession> {
  await page.addInitScript(INIT);
  const cdp = await page.context().newCDPSession(page);
  await cdp.send("Performance.enable");
  if (RALENTI > 1) await cdp.send("Emulation.setCPUThrottlingRate", { rate: RALENTI });
  let n = 0;
  page.on("request", (r) => { if (/firestore\.googleapis\.com/.test(r.url())) n++; });
  (page as any).__fs = () => n;
  return cdp;
}

async function metriques(cdp: CDPSession) {
  const { metrics } = await cdp.send("Performance.getMetrics");
  const m: Record<string, number> = {};
  for (const x of metrics) m[x.name] = x.value;
  return m;
}

/** Attend que le DOM et les tâches longues se taisent `calme` ms (borne : `max`). */
async function attendreCalme(page: Page, calme = 500, max = 30_000) {
  return page.evaluate(({ calme, max }) => new Promise<{ fin: number; borne: boolean }>((res) => {
    const w = window as unknown as Record<string, any>;
    const debut = performance.now();
    const tic = () => {
      const fin = Math.max(w.__m.lastMut, w.__m.lastLong);
      const now = performance.now();
      if (now - fin >= calme) return res({ fin, borne: false });
      if (now - debut > max) return res({ fin, borne: true });
      setTimeout(tic, 50);
    };
    requestAnimationFrame(() => requestAnimationFrame(tic));
  }), { calme, max });
}

export type Res = { ecran: string; action: string; ms: number | null; long: number; script: number; layout: number; style: number; fs: number; note?: string };

/** Mesure une action (un geste) : du premier pointerdown/keydown à la dernière mutation ou tâche longue. */
export async function mesure(page: Page, cdp: CDPSession, ecran: string, action: string, geste: () => Promise<unknown>, pret?: () => Promise<unknown>, note?: string): Promise<Res> {
  try {
    await page.evaluate(() => { const w = window as unknown as Record<string, any>; w.__m.t0 = 0; w.__m.long = 0; w.__m.nLong = 0; w.__m.lastLong = 0; w.__m.lastMut = 0; w.__m.arme = true; });
    const fs0 = (page as any).__fs();
    const m0 = await metriques(cdp);
    const tJs = Date.now();
    await geste();
    if (pret) await pret();
    const c = await attendreCalme(page);
    const m1 = await metriques(cdp);
    const r = await page.evaluate(() => { const w = window as unknown as Record<string, any>; w.__m.arme = false; return { t0: w.__m.t0, long: w.__m.long }; });
    const ms = r.t0 ? Math.max(0, Math.round(c.fin - r.t0)) : Date.now() - tJs;
    const d = (k: string) => Math.round(((m1[k] ?? 0) - (m0[k] ?? 0)) * 1000);
    return ecrire({ ecran, action, ms, long: Math.round(r.long), script: d("ScriptDuration"), layout: d("LayoutDuration"), style: d("RecalcStyleDuration"), fs: (page as any).__fs() - fs0, note: [note, c.borne ? "jamais calme (borne 30 s)" : "", r.t0 ? "" : "sans geste"].filter(Boolean).join("; ") || undefined });
  } catch (e) {
    return ecrire({ ecran, action, ms: null, long: 0, script: 0, layout: 0, style: 0, fs: 0, note: "échec : " + String(e).split("\n")[0].slice(0, 160) });
  }
}

/** Ouverture d'une adresse (chargement complet) : de la navigation au calme. */
export async function ouvrir(page: Page, cdp: CDPSession, ecran: string, url: string, pret?: () => Promise<unknown>): Promise<Res> {
  try {
    const fs0 = (page as any).__fs();
    const m0 = await metriques(cdp);
    await page.goto(url);
    if (pret) await pret();
    const c = await attendreCalme(page);
    const m1 = await metriques(cdp);
    const r = await page.evaluate(() => {
      const w = window as unknown as Record<string, any>;
      const js = performance.getEntriesByType("resource").filter((e) => /\.js(\?|$)/.test(e.name)) as PerformanceResourceTiming[];
      return { long: w.__m.long, js: js.reduce((s, e) => s + (e.encodedBodySize || 0), 0), jsBrut: js.reduce((s, e) => s + (e.decodedBodySize || 0), 0), nJs: js.length };
    });
    const d = (k: string) => Math.round(((m1[k] ?? 0) - (m0[k] ?? 0)) * 1000);
    return ecrire({ ecran, action: "ouverture " + url, ms: Math.round(c.fin), long: Math.round(r.long), script: d("ScriptDuration"), layout: d("LayoutDuration"), style: d("RecalcStyleDuration"), fs: (page as any).__fs() - fs0, note: `JS ${Math.round(r.js / 1024)} Ko compressé, ${Math.round(r.jsBrut / 1024)} Ko brut, ${r.nJs} fichiers` + (c.borne ? "; jamais calme" : "") });
  } catch (e) {
    return ecrire({ ecran, action: "ouverture " + url, ms: null, long: 0, script: 0, layout: 0, style: 0, fs: 0, note: "échec : " + String(e).split("\n")[0].slice(0, 160) });
  }
}

function ecrire(r: Res): Res {
  const ligne = { ...r, largeur: LARGEUR, cpu: RALENTI, mode: MODE, quand: new Date().toISOString() };
  appendFileSync(SORTIE, JSON.stringify(ligne) + "\n");
  console.log(`[${LARGEUR}] ${r.ecran} | ${r.action} | ${r.ms} ms | long ${r.long} | js ${r.script} lay ${r.layout} sty ${r.style} | fs ${r.fs}${r.note ? " | " + r.note : ""}`);
  return r;
}

/** Premier élément visible d'un locator, ou null après `ms`. */
export async function visible(l: Locator, ms = 4000): Promise<Locator | null> {
  const v = l.filter({ visible: true }).first();
  try { await v.waitFor({ timeout: ms }); return v; } catch { return null; }
}
